如果有 $ACCOUNT，命令成功，执行 then 后面的。如果没有 $ACCOUNT，命令失败，不执行 then 后面的，跳去 else 或 fi
if getent passwd "$ACCOUNT" >/dev/null 2>&1; then
    line=$(getent passwd "$ACCOUNT")   #把查到的用户信息保存到变量 line 里
    uid=$(printf '%s' "$line" | cut -d: -f3)   #把 line 的内容原样输出，用管道交给 cut，cut 按冒号切开，取第 3 段，然后把结果存到变量 uid
    pgid=$(printf '%s' "$line" | cut -d: -f4)    #同上
    pshell=$(printf '%s' "$line" | cut -d: -f7)   #同上
    case "$uid" in ''|*[!0-9]*) die "已有账号 $ACCOUNT 的 UID 非法: $uid" ;; esac #如果 $uid 是空的，或者包含非数字字符，就执行 die，报错退出
    [ "$uid" -ge 200 ] && [ "$uid" -le 999 ] || die "已有账号 UID=$uid 不在 200-999" #如果UID不为 200–999就报错退出
    [ "$pgid" = "$GID" ]       || die "已有账号的主组不是 $GROUP"
    [ "$pshell" = "$NOLOGIN" ] || die "已有账号的 shell 不是 $NOLOGIN"
else
    # 系统账号默认范围可能包含 100-199，不满足要求 → 自己从 200 找空号
    newuid=       #定义一个变量 newuid，先设为空
    for u in $(seq 200 999); do      #seq 200 999：生成 200 到 999 的数字序列
        if ! getent passwd "$u" >/dev/null 2>&1; then newuid=$u; break; fi #从 200 开始，找到第一个没人用的 UID，就记下来，并停止循环
    done
    [ -n "$newuid" ] || die "200-999 没有空闲 UID"  #如果 newuid 不是空的，说明找到了空闲 UID(因为一开始设置了newuid为空).如果 newuid 是空的，说明 200–999 全被占用了，报错退出
    useradd --system --no-create-home --home-dir /nonexistent \
            --shell "$NOLOGIN" --gid "$GROUP" --uid "$newuid" "$ACCOUNT" \  
            #创建用户的命令/创建系统账号/不创建 home 目录/home 目录设为 /nonexistent/  登录 shell 设为 /usr/sbin/nologin  /主组设为 archive/UID 设为刚找到的空闲号/账号名是 archiver
        || die "创建账号 $ACCOUNT 失败"
fi

常用权限数字
数字	字符	含义
644	rw-r--r--	属主读写，其他人只读
640	rw-r-----	属主读写，同组只读，其他人无权

stat是Linux 里查看文件元信息的命令（inode、大小、权限、时间戳等）

在-c '%a'里面-c 是 --format 的简写，指定输出格式。%a 表示八进制权限位，就是 ls -l 里那串 rwx 对应的数字

[ "$mode" = "640" ]判断权限是不是 640

-c 表示：
    按我指定的格式输出，不要输出默认那一大堆。
'%U:%G' 是格式：
    %U：属主名，也就是 User name
    :：冒号，分隔符
    %G：属组名，也就是 Group name

chown 是 change owner 的缩写，用来修改文件或目录的所有者（属主）和所属组（属组）.基本语法：chown [选项] 用户[:组] 文件...

# ---------- 3. 文件归属与权限 ----------
# 先 chmod 再 chown：chmod 需要"属主身份"，此刻 root 还是属主；
# 一旦 chown 给 archiver，root 就不是属主了。本关 root 能力被削，一旦 chown 走，root 连 chmod 都做不了了，所以顺序不能反
mode=$(stat -c '%a' "$TASK_FILE")
[ "$mode" = "640" ] || chmod 640 "$TASK_FILE" || die "chmod 失败"  #先看权限是不是 640。如果是，什么都不做，继续。如果不是，就执行 chmod 640。如果 chmod 也失败，就报错退出
owner=$(stat -c '%U:%G' "$TASK_FILE")   #用 stat 查看 $TASK_FILE 的属主和属组，按 属主:属组 的格式输出，然后把结果存到变量 owner
[ "$owner" = "$ACCOUNT:$GROUP" ] || chown "$ACCOUNT:$GROUP" "$TASK_FILE" || die "chown 失败"

先看文件属主属组是不是 archiver:archive。如果是，什么都不做，继续。如果不是，就执行 chown archiver:archive。如果 chown 也失败，就报错退出

&& 意思是：如果前面的命令成功，才执行后面的命令。

runuser 是切换用户执行命令的工具,格式：runuser -u 用户名 -- 命令

su 是切换用户的命令,基本格式：su -s shell -c "命令" 用户名

runuser：更现代，专门给 root 用来切用户

su：更老、更通用，几乎每个 Linux 都有

这里两个都写，是为了多一种尝试：万一 runuser 不可用，还有 su 兜底

为什么不用默认 shell?因为 archiver 的登录 shell 是：/usr/sbin/nologin  ,nologin 是不允许登录的 shell。如果直接 su archiver，可能因为 nologin 而失败。所以这里强制指定/bin/sh

chmod = change mode，改的是权限,也就是谁能读、谁能写、谁能执行

chown = change owner，改的是属主和属组, 格式是 chown 属主:属组 文件

# ---------- 4. 读任务说明 ----------
# 本关 root 没有 CAP_DAC_OVERRIDE，读不了 640 的属主文件。
# 依次尝试：直接读 → 换成 archiver 身份读 → 临时交回 root 读。
read_task() {
    cat "$TASK_FILE" 2>/dev/null && return 0   #可以直接读就return 0,不行就继续下面的步骤
    runuser -u "$ACCOUNT" -- cat "$TASK_FILE" 2>/dev/null && return 0   #以 archiver 用户的身份，运行 cat /workspace/task.txt
    su -s /bin/sh -c "cat '$TASK_FILE'" "$ACCOUNT" 2>/dev/null && return 0   #切换到 archiver 用户，用 /bin/sh 执行 cat '/workspace/task.txt' 成功就退出
    chown root:root "$TASK_FILE" 2>/dev/null || return 1   #试着把任务文件改回 root:root。如果改不了，函数就返回失败，后面不试了。如果改成功，继续往下走
    chmod 600 "$TASK_FILE" 2>/dev/null   #临时把文件交给 root 时，只让 root 自己能读，别让同组的人也能读.更安全.
    cat "$TASK_FILE" 2>/dev/null
    rc=$?
    chmod 640 "$TASK_FILE" 2>/dev/null   #把文件权限改成 640
    chown "$ACCOUNT:$GROUP" "$TASK_FILE" 2>/dev/null   #把 $TASK_FILE 的属主改成 $ACCOUNT，属组改成 $GROUP
    return $rc
}

content=$(read_task) || die "无法读取 $TASK_FILE"
